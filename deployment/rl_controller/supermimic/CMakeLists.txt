cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(supermimic_deployment LANGUAGES CXX)
  set(SUPERMIMIC_STANDALONE ON)
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(SM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  set(SM_ARCH aarch64)
else()
  set(SM_ARCH x86_64)
endif()
set(SM_ORT "${SM_ROOT}/third_party/onnxruntime-${SM_ARCH}")
add_library(supermimic_core core.cpp)
target_include_directories(supermimic_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${SM_ORT}/include")
if(TARGET yaml-cpp::yaml-cpp)
  set(SM_YAML_TARGET yaml-cpp::yaml-cpp)
else()
  set(SM_YAML_TARGET yaml-cpp)
endif()
target_link_libraries(supermimic_core PUBLIC Eigen3::Eigen ${SM_YAML_TARGET} "${SM_ORT}/lib/libonnxruntime.so" Threads::Threads)
target_compile_options(supermimic_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(supermimic_probe probe.cpp)
target_link_libraries(supermimic_probe supermimic_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(supermimic_live_switch_probe "${SM_ROOT}/tests/supermimic_live_switch_probe.cpp")
  target_link_libraries(supermimic_live_switch_probe supermimic_core)
  add_executable(supermimic_live_switch_check "${SM_ROOT}/tests/supermimic_live_switch_check.cpp")
  target_link_libraries(supermimic_live_switch_check supermimic_core)
  add_test(NAME supermimic_live_switch_check COMMAND supermimic_live_switch_check
    "${SM_ROOT}/config/model/supermimic_walk1_subject2_proprio"
    "${SM_ROOT}/config/model/supermimic_motion_bank/dance1_subject2")
  add_executable(supermimic_reference_hold_check "${SM_ROOT}/tests/supermimic_reference_hold_check.cpp")
  target_link_libraries(supermimic_reference_hold_check supermimic_core)
  add_test(NAME supermimic_reference_hold_check COMMAND supermimic_reference_hold_check
    "${SM_ROOT}/config/model/supermimic_walk1_subject2_proprio")
  add_executable(supermimic_joystick_check "${SM_ROOT}/tests/supermimic_joystick_check.cpp")
  add_test(NAME supermimic_joystick_check COMMAND supermimic_joystick_check)
  add_executable(supermimic_motion_bank_check "${SM_ROOT}/tests/supermimic_motion_bank_check.cpp")
  target_link_libraries(supermimic_motion_bank_check supermimic_core)
  add_test(NAME supermimic_motion_bank_check COMMAND supermimic_motion_bank_check
    "${SM_ROOT}/config/model/supermimic_walk1_subject2_proprio"
    "${SM_ROOT}/config/model/supermimic_motion_bank")
endif()
if(SUPERMIMIC_STANDALONE)
  option(SUPERMIMIC_BUILD_ROS "Build ROS hardware runner" OFF)
else()
  option(SUPERMIMIC_BUILD_ROS "Build ROS hardware runner" ON)
endif()
if(SUPERMIMIC_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rcl_interfaces REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  find_package(crb_ros_msg REQUIRED)
  find_package(rmw REQUIRED)
  add_executable(supermimic_runner runner.cpp)
  target_link_libraries(supermimic_runner supermimic_core)
  ament_target_dependencies(supermimic_runner rclcpp rcl_interfaces sensor_msgs geometry_msgs std_msgs std_srvs crb_ros_msg)
  add_executable(supermimic_sensor_probe sensor_probe.cpp)
  ament_target_dependencies(supermimic_sensor_probe rclcpp rmw sensor_msgs std_msgs crb_ros_msg)
  set_target_properties(supermimic_sensor_probe PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  install(TARGETS supermimic_runner supermimic_probe supermimic_sensor_probe RUNTIME DESTINATION bin)
  set_target_properties(supermimic_runner supermimic_probe PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  if(SUPERMIMIC_STANDALONE)
    install(FILES "${SM_ROOT}/config/supermimic_walk4.yaml" "${SM_ROOT}/config/supermimic_walk4_proprio.yaml" "${SM_ROOT}/config/supermimic_walk1_subject2_proprio.yaml" DESTINATION config)
    install(FILES "${SM_ROOT}/config/supermimic_udp_only.xml" DESTINATION config)
    install(DIRECTORY "${SM_ROOT}/config/model/supermimic_walk4" DESTINATION config/model)
    install(DIRECTORY "${SM_ROOT}/config/model/supermimic_walk4_proprio" DESTINATION config/model OPTIONAL)
    install(DIRECTORY "${SM_ROOT}/config/model/supermimic_walk1_subject2_proprio" DESTINATION config/model OPTIONAL)
    install(DIRECTORY "${SM_ROOT}/config/model/supermimic_motion_bank" DESTINATION config/model OPTIONAL)
    file(GLOB SM_RUNTIME_LIBS "${SM_ORT}/lib/libonnxruntime.so*")
    install(FILES ${SM_RUNTIME_LIBS} DESTINATION lib)
    get_target_property(SM_YAML_TYPE ${SM_YAML_TARGET} TYPE)
    if(SM_YAML_TYPE STREQUAL "SHARED_LIBRARY")
      install(FILES "$<TARGET_FILE:${SM_YAML_TARGET}>" "$<TARGET_SONAME_FILE:${SM_YAML_TARGET}>" DESTINATION lib)
    endif()
    file(GLOB SM_MESSAGE_LIBS "${crb_ros_msg_DIR}/../../../lib/libcrb_ros_msg*.so*")
    install(FILES ${SM_MESSAGE_LIBS} DESTINATION lib)
    install(PROGRAMS "${SM_ROOT}/scripts/run_supermimic.sh" DESTINATION scripts)
    install(PROGRAMS "${SM_ROOT}/scripts/supermimic_motion.py" DESTINATION scripts)
    install(PROGRAMS "${SM_ROOT}/scripts/diagnose_supermimic_sensors.sh" DESTINATION scripts)
  endif()
endif()
